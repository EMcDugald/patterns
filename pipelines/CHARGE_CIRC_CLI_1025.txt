python pgb_charge_circ_survey.py --npz results/1011_lastframes_mu65_92 --outdir results/1025_charge_circ_mu65_92

python pgb_charge_circ_survey.py --npz "$(find . -path '*energy_vs_angle_zigzag_0929_np4_full*' -name '*.npz' | grep -i cache | sort | paste -sd, -)" --outdir results/1025_charge_circ_energy_sweep

# 4. packages for the chat: all text in one zip; the set figures and the t = 200 per-angle figures in another
zip -r results/1025_charge_circ_text.zip results/1025_charge_circ_* -i "*.txt" "*.csv" "*.md" "*.json"
